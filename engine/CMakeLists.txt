cmake_minimum_required(VERSION 3.24)
project(PrivexStudioEngine VERSION 0.2.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
if(NOT DEFINED OBS_SOURCE_DIR)
  message(FATAL_ERROR "OBS_SOURCE_DIR is required")
endif()
set(STUDIO_VERSION "${PROJECT_VERSION}" CACHE STRING "Privex Studio release version")
if(NOT STUDIO_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+(-[A-Za-z0-9.-]+)?$")
  message(FATAL_ERROR "Invalid STUDIO_VERSION")
endif()
string(REGEX MATCH "^([0-9]+)\\.([0-9]+)\\.([0-9]+)" _version_parts "${STUDIO_VERSION}")
set(STUDIO_VERSION_MAJOR "${CMAKE_MATCH_1}")
set(STUDIO_VERSION_MINOR "${CMAKE_MATCH_2}")
set(STUDIO_VERSION_PATCH "${CMAKE_MATCH_3}")
configure_file(version.rc.in version.rc @ONLY)
add_executable(PrivexStudioEngine main.cpp "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
target_include_directories(PrivexStudioEngine PRIVATE "${OBS_SOURCE_DIR}/libobs" "${OBS_SOURCE_DIR}/build_x64/config")
target_link_libraries(PrivexStudioEngine PRIVATE Qt6::Core "${OBS_SOURCE_DIR}/build_x64/libobs/RelWithDebInfo/obs.lib" user32 ole32)
target_compile_definitions(PrivexStudioEngine PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(PrivexStudioEngine PRIVATE /W4 /utf-8 /guard:cf)
target_link_options(PrivexStudioEngine PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
